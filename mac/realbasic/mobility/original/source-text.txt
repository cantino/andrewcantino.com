joint.shift:
Sub shift(s as double)
  dim amount,slope,midx,midy,distperslope,amount,newamount as double
  
  ' The following is the slope of the line between the two known points.
  slope = velocity.getSlope
  
  ' Hopefully the distance traveled per slope step
  distperslope = sqrt((slope*slope) + 1)
  
  ' The following is the amount to shift up or down from the midpoint.
  amount = velocity.mag * s
     
  ' This is the number of 'slopes' needed to be traveled
  newamount = amount/distperslope
  
  if (slope > 0) then
    if (velocity.angle < 180) then
      'going up!
      x = x + newamount
      y = y + newamount*slope
    else
      'going down
      x = x - newamount
      y = y - newamount*slope
    end if
  else
    if (velocity.angle < 180) then
      'going up!
      x = x - newamount
      y = y - newamount*slope
    else
      'going down
      x = x + newamount
      y = y + newamount*slope
    end if
  end if
  
  if (y <= floorY) then
    if (velocity.angle > 180 and velocity.angle <= 270) then
      velocity.angle = 90 + (90 - (velocity.angle - 180))
    elseif (velocity.angle > 270 and velocity.angle < 360) then
      velocity.angle = 90 - ((velocity.angle - 180) - 90)
    end if
    'if (velocity.mag < 5) then
    y = y + abs(floorY - y)*.5
    'end if
    velocity.mag = velocity.mag * floorAbsorbtion
  end if
End Sub

vector.getSlope:
Function getSlope() As double
  return tan(angle*(PI/180))
End Function

vector.setSlope:
Sub setSlope(slope as double)
  angle = atan(slope)*(180/PI)
  if (angle < 0) then
    angle = 360 + angle
  end if
End Sub

vector.trueSetSlope:
Sub trueSetSlope(y as double, x as double)
  if (x <= 0 and y <= 0) then
    angle = 180 + (atan(y/x)*(180/PI))
  elseif (x < 0 and y >= 0) then
    angle = 180 + (atan(y/x)*(180/PI))
  elseif (x >= 0 and y < 0) then
    angle = 360 + (atan(y/x)*(180/PI))
  else
    angle = atan(y/x)*(180/PI)
  end if
  if (angle < 0) then
    angle = 360 + angle
  end if
End Sub

vector.mirror:
Sub mirror(v as vector)
  angle = v.angle
  mag = v.mag
End Sub

MathAndMethods.userLoadSavedFramework:
Function userLoadSavedFramework() As Framework
  dim f as folderitem
  dim fw as Framework
  dim s as textinputstream
  dim ipt as string
  
  fw = new Framework
  
  f = getopenfolderitem("text")
  
  if (f <> nil and f.exists) then
    s = f.openastextfile
    if (s <> nil) then
      
      while (s.eof = false)
        ipt = s.readline
        Select Case nthfield(ipt," ",1)
        Case "JT"
          fw.addJoint(val(nthfield(ipt," ",2)),val(nthfield(ipt," ",3)))
          fw.joints(ubound(fw.joints)).mass = val(nthfield(ipt," ",4))
          if (fw.joints(ubound(fw.joints)).mass = 0) then
            fw.joints(ubound(fw.joints)).mass = 1
          end if
          if (nthfield(ipt," ",5) = "true") then
            fw.joints(ubound(fw.joints)).locked = true
          else
            fw.joints(ubound(fw.joints)).locked = false
          end if
          fw.joints(ubound(fw.joints)).velocity.angle = val(nthfield(ipt," ",6))
          fw.joints(ubound(fw.joints)).velocity.mag = val(nthfield(ipt," ",7))
        Case "SP"
          fw.addSpring(val(nthfield(ipt," ",2)),val(nthfield(ipt," ",3)))
          fw.springs(ubound(fw.springs)).wavePosition = val(nthfield(ipt," ",4))
        Case "WF"
          fw.addWaveForm(val(nthfield(ipt," ",2)),val(nthfield(ipt," ",3)),val(nthfield(ipt," ",4)),val(nthfield(ipt," ",5)))
        Else
          msgbox "'" + ipt + "' is not a valid definition line."
        End Select
      wend
      
      s.close
      fw.doStart
      return fw
      
    end if
  end if
End Function

    if (not folder.exists) then

          s.writeline "JT " + str(f.joints(a).x) + " " + str(f.joints(a).y) + " " + str(f.joints(a).mass) + " " + l
        next
        
        for a = 1 to ubound(f.springs)
          s.writeline "SP " + str(f.springs(a).Jt1Code) + " " + str(f.springs(a).Jt2Code) + " " + str(f.springs(a).wavePosition)
        next
        
        for a = 1 to ubound(f.waveforms)
          s.writeline "WF " + str(f.waveforms(a).average) + " " + str(f.waveforms(a).amplitude) + " " + str(f.waveforms(a).period) + " " + str(f.waveforms(a).phase)
        next
        
        s.close
        folder.mactype = "TEXT"
        folder.maccreator = "ljwt"
      end if
    end if
  end if
End Sub

Function dist(p1 as integer, p2 as integer) As Double
  return sqrt((joints(p2).y - joints(p1).y)*(joints(p2).y - joints(p1).y) + (joints(p2).x - joints(p1).x)*(joints(p2).x - joints(p1).x))
End Function

Framework.tick:
Sub tick(s as double)
  dim a,b,c as integer
  dim slope,force,xcomponent,ycomponent,xsum,ysum as double
  dim finalvolx,finalvoly as double
  dim v as vector
  
  ' Okay, first we find all of the vectors for every point, then we
  ' find the sums for every point and move the points accordingly.
  ' This model should account for floor friction, gravity, and springs.
  
  ' The following function increments the waveforms.  It will also
  ' update the spring restLengths.
  tickWaveforms(waveFormSpeed)
  
  ' The following function adds force vectors for all springs in the springs array.
  addSpringVectors
  
  ' Now we find vector sums for every joint and shift the points the nessesary
  ' amount.
  
  for a = 1 to ubound(joints)
    if (not joints(a).locked) then
      xsum = 0
      ysum = 0
      for b = 1 to ubound(joints(a).vectors)
        xcomponent = joints(a).vectors(b).mag * cos(joints(a).vectors(b).angle*(PI/180))
        ycomponent = joints(a).vectors(b).mag * sin(joints(a).vectors(b).angle*(PI/180))
        xsum = xsum + xcomponent
        ysum = ysum + ycomponent
        if (debuging) then
          o.add("Joint " + str(a) + ", vector " + str(b) + ":  xcomponent = " + str(xcomponent) + "  ycomponent = " + str(ycomponent) + "  xsum = " + str(xsum) + "  ysum = " + str(ysum) + chr(13))
        end if
      next
      
      ' Lets check for floor friction.  This friction is an approximation.  It DOES
      ' NOT take the previous momentum into account!
      
      if (joints(a).y <= (floorY+2)) then
        ' Okay, we are on the floor.  Any x velocity should have friction 
        ' applied to it.
        
        if (abs(floorFrict*ysum) > abs(xsum)) then
          if (debuging) then
            o.add("Joint " + str(a) + ": Friction applied (to zero, old velx = " + str((joints(a).velocity.mag * cos(joints(a).velocity.angle * (PI/180)))) + ")." + chr(13))
          end if
          xsum = .00000001
        else
          if (xsum > 0) then
            xsum = xsum - abs(floorFrict*ysum)
          else
            xsum = xsum + abs(floorFrict*ysum)
          end if
          if (debuging) then
            o.add("Joint " + str(a) + ": Friction applied.  (Amt. left: " + str(xsum) + ")" + chr(13))
          end if
        end if
      end if
      
      ' Fx = MAx, Fy = MAy
      
      finalvolx = (joints(a).velocity.mag * cos(joints(a).velocity.angle * (PI/180)))*airFriction + ((xsum/joints(a).mass)*s)
      finalvoly = (joints(a).velocity.mag * sin(joints(a).velocity.angle * (PI/180)))*airFriction + ((ysum/joints(a).mass)*s)
      
      if (debuging) then
        o.add("Joint " + str(a) + " Finalvolx: " + str(finalvolx) + " Finalvoly: " + str(finalvoly) + chr(13))
      end if
      
      joints(a).velocity.trueSetSlope(finalvoly,finalvolx)
      
      joints(a).velocity.mag = sqrt((finalvoly*finalvoly) + (finalvolx*finalvolx))
      
      if (joints(a).velocity.angle < 0) then
        joints(a).velocity.angle = 360 + joints(a).velocity.angle
      end if
      
      while (joints(a).velocity.angle > 360)
        joints(a).velocity.angle = joints(a).velocity.angle - 360
      wend
      
      joints(a).shift(s)
    end if
  next
  
  if (debuging) then
    for a = 1 to ubound(joints)
      o.add("Joint " + str(a) + chr(13) + joints(a).debug + chr(13))
    next
  end if
  
  ' Lets keep track of the current 'time'
  time = time + s
  
  ' Lets reset everything.
  for a = 1 to ubound(joints)
    joints(a).joint
  next
End Sub

Framework.addSpringVectors:
Sub addSpringVectors()
  dim a as integer
  dim slope,force,newRestLength as double
  dim v as vector
  
  ' The following loop adds force vectors for all springs in the springs array.
  for a = 1 to ubound(springs)
    v = new vector
    
    if (springs(a) = nil) then
      msgbox "Spring is nil!"
    elseif (joints(springs(a).Jt2Code) = nil) then
      msgbox "joints(springs(a).Jt2Code) is nil!"
    elseif (joints(springs(a).Jt1Code) = nil) then
      msgbox "joints(springs(a).Jt1Code) is nil!"
    end if
    
    ' This is the slope between the two joints on the ends of the spring:
    slope = (joints(springs(a).Jt2Code).y - joints(springs(a).Jt1Code).y)/(joints(springs(a).Jt2Code).x - joints(springs(a).Jt1Code).x)
    
    ' Useful equations:
    '   Hooke's law: SpringForce = -kx, where x is the displacement
    '   Impulse: Force * time = mass * change in v
    '   Acceleration by a spring: -(k/m)*x
    
    ' This spring could be a muscle.  Lets find out if it is.
    newRestLength = springs(a).restLength
    
    ' Okay, lets look at the length of the spring and figure out about the force.
    force = -1 * springs(a).k * (dist(springs(a).Jt1Code,springs(a).Jt2Code) - newRestLength)
    
    ' If the force is negitive, the spring has been stretched too far.
    ' Both joints have vectors pointing towards the other joint.
    ' Otherwise, both joints have vectors pointing away from the other joint.
    
    ' if top, and neg force, vector is down
    ' if top and pos force, vector is up
    ' if bottom and neg force, vector is up
    ' if bottom and pos force, vector is down
    
    ' if the slope is positive, to go down, the angle must be 180 + atan(slope)*(180/PI)
    ' if the slope is positive, to go up, the angle must be atan(slope)*(180/PI)
    ' if the slope is neg, to go down, the angle must be 360 + atan(slope)*(180/PI)
    ' if the slope is neg, to go up, the angle must be 180 + atan(slope)*(180/PI)
    
    if (debuging) then
      o.add(str(a) + "  Slope: " + str(slope) + "  Force: " + str(force) + chr(13))
    end if
    
    
    if (joints(springs(a).Jt1Code).y >= joints(springs(a).Jt2Code).y) then
      ' top
      if (force < 0) then
        'neg force
        if (slope > 0) then
          'pos slope
          v.angle = 180 + atan(slope)*(180/PI)
        else
          'neg slope
          v.angle = 360 + atan(slope)*(180/PI)
        end if
      else
        'pos force
        if (slope > 0) then
          'pos slope
          v.angle = atan(slope)*(180/PI)
        else
          'neg slope
          v.angle = 180 + atan(slope)*(180/PI)
        end if
      end if
    else
      'bottom
      if (force > 0) then
        'pos force
        if (slope > 0) then
          'pos slope
          v.angle = 180 + atan(slope)*(180/PI)
        else
          'neg slope
          v.angle = 360 + atan(slope)*(180/PI)
        end if
      else
        'neg force
        if (slope > 0) then
          'pos slope
          v.angle = atan(slope)*(180/PI)
        else
          'neg slope
          v.angle = 180 + atan(slope)*(180/PI)
        end if
      end if
    end if
    v.mag = abs(force)*absorbtion
    joints(springs(a).Jt1Code).addVector v
    
    v.mag = abs(force)*absorbtion
    if (v.angle >= 0 and v.angle <= 180) then
      v.angle = v.angle + 180
    else
      v.angle = v.angle - 180
    end if
    
    if (v.angle < 0) then
      v.angle = 360 + v.angle   'Make sure this shouldn't be a - sign
    elseif (v.angle > 360) then
      v.angle = v.angle - 360
    end if
    joints(springs(a).Jt2Code).addVector v
  next
  
Exception err as OutOfBoundsException
  msgbox "Ubound of springs: " + str(ubound(springs))
  msgbox "Jt1Code of a is " + str(springs(a).Jt1Code)
  msgbox "Jt2Code of a is " + str(springs(a).Jt2Code)
  msgbox "Ubound of joints: " + str(ubound(joints))
  MsgBox "Out of bounds!"
Exception err as NilObjectException
  MsgBox "Tried to access a Nil object!"
End Sub

Framework.paintWave:
Function paintWave(w as integer, h as integer) As picture
  dim p as picture
  dim a,b as integer
  dim c as double
  p = newpicture(w,h,4)
  
  p.graphics.forecolor = rgb(0,0,0)
  
  for a = 1 to w
    c = 0
    for b = 1 to ubound(waveforms)

Function at(x as double) As Double
  ' The general formula for a sinusoidal oscillation:
  ' A + Bcos[ 2PI

PrimeModule.getPrimes:
Sub getPrimes(GenTo as integer)
  dim b,hold,a as integer
  redim primes(2)
  a = 5
  primes(0) = 2
  primes(1) = 3
  primes(2) = 5
  while not usercancelled and ubound(primes) < GenTo
    a = a + 1
    hold = 0
    for b = 0 to ubound(primes)-1
      if (a = primes(b)) then
        hold = 1
        b = ubound(primes)
      elseif (floor(a/primes(b)) = a/primes(b)) then
        hold = 1
        goto out
      end if
    next
    out:
    if (hold = 0) then
      primes.append a
    end if
  wend
End Sub

PrimeModule.isPrime:
Function isPrime(n as integer) As boolean
  dim b as integer
  if primes(ubound(primes)) < ((n+1)/2) then
    msgbox "Error, not enough primes known."
    return false
  else
    for b = 0 to ubound(primes)-2
      if (n = primes(b)) then
        'return false
      elseif (floor(n/primes(b)) = n/primes(b)) then
        return false
      end if
    next
    return true
  end if
End Function

PrimeModule.primeFactors:
Function primeFactors(n as integer) As list
  dim a,b,c as integer
  dim bool as boolean
  dim l as list
  l = new list
  
  bool = isPrime(n)
  
  a = n
  b = 0
  
  while (a > 1)
    if ((a/primes(b)) = floor(a/primes(b))) then
      
      l.append primes(b)
      a = a/primes(b)
      
    else
      
      b = b + 1
      
    end if
  wend
  
  return l
End Function

PrimeModule.lcm:
Function lcm(n1 as integer, n2 as integer) As integer
  dim l1, l2,temp as list
  dim a,b,c as integer
  
  ' This method does not seem to be quite as effective as the
  ' bruteLCm module.  However, for large numbers is it MUCH faster.
  
  b = 1
  temp = new list
  
  l1 = primeFactors(n1)
  l2 = primeFactors(n2)
  
  for a = 1 to ubound(l1.list)
    if (l1.numContains(l1.get(a)) > l2.numContains(l1.get(a))) then
      temp.append l1.get(a)
    end if
  next
  
  for a = 1 to ubound(l2.list)
    temp.append l2.get(a)
  next
  
  for a = 1 to ubound(temp.list)
    b = b * temp.get(a)
  next
  
  return b
  
End Function

PrimeModule.bruteLCM:
Function bruteLCM(n1 as integer, n2 as integer) As integer
  dim a,b,c as integer
  
  do
    b = b + 1
    if (n1 > n2) then
      a = n1 * b
    else
      a = n2 * b
    end if
  loop until (((a/n1) = floor(a/n1)) and ((a/n2) = floor(a/n2)))
  
  return a
End Function

  for a = 1 to spawnNumber
    frames.append new Framework
    frames(ubound(frames)).mirror(frames(1))
    choice = round(rnd*5)+1
    
    if (choice = 1) then
      ' We will mutate the waveforms
      addlog("Mutating waveform" + chr(13))
      mutateWaveform
    elseif (choice = 2) then
      ' We will mutate existing positions on the waveform or
      ' we will add a new position
      addlog("Mutating waveform positions" + chr(13))
      mutateWaveformPositions
    elseif (choice = 3) then
      ' We will add a new joint
      b = round(rnd*4)+1
      if (b < 4) then
        addlog("Adding a random joint between" + chr(13))
        addRandomJointBetween
      else
        addlog("Adding a random joint" + chr(13))
        addRandomJoint
      end if
    elseif (choice = 4) then
      ' We will add a new spring
      addlog("Adding a random spring" + chr(13))
      addRandomSpring
    elseif (choice = 5) then
      ' We will remove a spring
      addlog("Removing random spring" + chr(13))
      removeRandomSpring
    elseif (choice = 6) then
      ' We will move the initial position of a joint
      addlog("Moving a joint" + chr(13))
      moveJoint
    else
      msgbox "Illeagal mutation choice: " + str(choice)
    end if
    
    frames(ubound(frames)).doStart
  next

MainWindow.addRandomJoint:
Sub addRandomJoint()
  dim x,y,a,newx,newy,ref,abort as integer
  dim rangle,slope,distperslope,newamount as double
  
  x = round(rnd*99) + 1
  y = round(rnd*99) + 1
  abort = 0
  while (y > -1*x + 100 and abort < 100)
    abort = abort + 1
    x = round(rnd*99) + 1
    y = round(rnd*99) + 1
  wend
  
  ' Okay, x is now a distance.  We use the probability curve: -1x + 100
  ' Now we need a random slope:
  
  rangle = round(rnd*359)+1
  slope = tan(rangle*(PI/180))
  
  ' Hopefully the distance traveled per slope step
  distperslope = sqrt((slope*slope) + 1)
  
  ' This is the number of 'slopes' needed to be traveled
  newamount = x/distperslope
  
  ref = round(rnd*(ubound(frames(ubound(frames)).joints)-1))+1
  
  if (ref > 0) then
    
    newx = frames(ubound(frames)).joints(ref).x
    newy = frames(ubound(frames)).joints(ref).y
    
    if (slope > 0) then
      if (rangle < 180) then
        'going up!
        newx = newx + newamount
        newy = newy + newamount*slope
      else
        'going down
        newx = newx - newamount
        newy = newy - newamount*slope
      end if
    else
      if (rangle < 180) then
        'going up!
        newx = newx - newamount
        newy = newy - newamount*slope
      else
        'going down
        newx = newx + newamount
        newy = newy + newamount*slope
      end if
    end if
    
    frames(ubound(frames)).addJoint(newx,newy)
    frames(ubound(frames)).addSpring(ref,ubound(frames(ubound(frames)).joints))
    
    addlog("Added random joint at " + str(newx) + "," + str(newy) + chr(13))
  end if
  
End Sub

        newx = newx - newamount
        newy = newy - newamount*slope
      end if
    else
      if (rangle < 180) then
        'going up!
        newx = newx - newamount
        newy = newy - newamount*slope
      else
        'going down
        newx = newx + newamount
        newy = newy + newamount*slope
      end if
    end if
    
    frames(ubound(frames)).joints(ref).x = newx
    frames(ubound(frames)).joints(ref).x = newy
    frames(ubound(frames)).updateAllSpringLengths
    
    addlog("Moved joint " + str(ref) + " to " + str(newx) + "," + str(newy) + chr(13))
  end if
  
End Sub

  pt1Unlinked = true
  pt2Unlinked = true

  if (ref > 0) then
    
    pt1 = frames(ubound(frames)).springs(ref).Jt1Code

    addlog("Removed spring " + str(ref))
    
    for a = 1 to ubound(frames(ubound(frames)).springs)
      if (frames(ubound(frames)).springs(a).Jt1Code = pt1 or frames(ubound(frames)).springs(a).Jt2Code = pt1) then
        pt1Unlinked = false
      end if
      if (frames(ubound(frames)).springs(a).Jt1Code = pt2 or frames(ubound(frames)).springs(a).Jt2Code = pt2) then
        pt2Unlinked = false
      end if
    next
    
    if (pt1Unlinked) then
      frames(ubound(frames)).joints.remove pt1
      addlog(" and joint " + str(pt1))
      for a = 1 to ubound(frames(ubound(frames)).springs)
        if (frames(ubound(frames)).springs(a).Jt1Code > pt1) then
          frames(ubound(frames)).springs(a).Jt1Code = frames(ubound(frames)).springs(a).Jt1Code - 1
        end if
        if (frames(ubound(frames)).springs(a).Jt2Code > pt1) then
          frames(ubound(frames)).springs(a).Jt2Code = frames(ubound(frames)).springs(a).Jt2Code - 1
        end if
      next
    end if
    if (pt2Unlinked) then
      frames(ubound(frames)).joints.remove pt2
      addlog(" and joint " + str(pt2))
      for a = 1 to ubound(frames(ubound(frames)).springs)
        if (frames(ubound(frames)).springs(a).Jt1Code > pt2) then
          frames(ubound(frames)).springs(a).Jt1Code = frames(ubound(frames)).springs(a).Jt1Code - 1
        end if
        if (frames(ubound(frames)).springs(a).Jt2Code > pt2) then
          frames(ubound(frames)).springs(a).Jt2Code = frames(ubound(frames)).springs(a).Jt2Code - 1
        end if
      next
    end if
  end if

end if

MainWindow.addRandomSpring:
Sub addRandomSpring()
  dim x,y,a,ref,bestJt,cycle,abort as integer
  dim dist,bestDist as double
  
  while (cycle < 10 and bestJt = 0)
    
    cycle = cycle + 1
    bestDist = 0
    bestJt = 0
    
    x = round(rnd*124) + 1
    y = round(rnd*124) + 1
    abort = 0
    while (y > -4*x + 500 and abort < 100)
      abort = abort + 1
      x = round(rnd*124) + 1
      y = round(rnd*124) + 1
    wend
    
    
    ref = round(rnd*(ubound(frames(ubound(frames)).joints)-1))+1
    
    if (ref > 0) then
      
      for a = 1 to ubound(frames(ubound(frames)).joints)
        dist = pow((frames(ubound(frames)).joints(ref).y - frames(ubound(frames)).joints(a).y),2)
        dist = dist + pow((frames(ubound(frames)).joints(ref).x - frames(ubound(frames)).joints(a).x),2)
        dist = sqrt(dist)
        if (dist > bestDist and dist < x) then
          bestDist = dist
          bestJt = a
        end if
      next
      
      if (bestJt > 0) then
        
        for a = 1 to ubound(frames(ubound(frames)).springs)
          if (frames(ubound(frames)).springs(a).Jt1Code = bestJt or frames(ubound(frames)).springs(a).Jt2Code = bestJt) then
            if (frames(ubound(frames)).springs(a).Jt1Code = ref or frames(ubound(frames)).springs(a).Jt2Code = ref) then
              goto exists
            end if
          end if
        next
        
        frames(ubound(frames)).addSpring(ref,bestJt)
        
        addlog("Added random spring from " + str(ref) + " to " + str(bestJt) + chr(13))
        
        exists:
        
      end if
    end if
    
  wend
End Sub
